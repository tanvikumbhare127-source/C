#include <iostream>
#include <cmath>
using namespace std;
int main() {
    int n, i, isPrime = 1;
    cout << "Enter a number: ";
    cin >> n;
    if (n < 2) isPrime = 0;
    else {
        for (i = 2; i <= sqrt(n); i++) {
            if (n % i == 0) {
            isPrime = 0;
            break;
        }
    }
}
 if (isPrime) cout << "Prime\n";
    else cout << "Not Prime\n";
    return 0;
}
